 #include <blitz/array.h>
 #include <fftw3.h>
 #include <complex>
 #include <cmath>
 #include <iostream>
 #include <cuda.h>
 #include <cufft.h>
 
 using namespace blitz;
 using std::complex;
 
 // Fill data with sum of two sine waves
 void fill_array(Array<float,2> &data) {
     for (int i=0; i < data.rows(); i++) {
         for (int j=0; j < data.cols(); j++) {
             float x = (float)i / 25.0f; 
             float y = (float)j / 10.0f; 
             data(i,j) = std::sin(2.0f * M_PI * x) + std::sin(2.0f * M_PI * y);
         }
     }
 }
 
 // Validate that forward transform "kdata" inverts to "data" within tolerance
 bool validate(Array<float,2> &data, Array<complex<float>, 2> kdata) {
     Array<float,2> rdata(data.extent());
     fftwf_plan plan = fftwf_plan_dft_c2r_2d(data.rows(), data.cols(),
         reinterpret_cast<fftwf_complex*>(kdata.data()), rdata.data(), FFTW_ESTIMATE);
     fftwf_execute(plan);
     fftwf_destroy_plan(plan);
     rdata /= data.size(); 
     return all(abs(data - rdata) < 1e-5f);
 }
 
 int main() {
     int n = 10000;
 
     //=============================================================
     // 1) CPU Out-of-place 2D R2C (FFTW)
     //=============================================================
     Array<float,2> rdata1(n,n);
     Array<complex<float>, 2> kdata1(n, n/2 + 1);
 
     // Plan
     fftwf_plan plan1  = fftwf_plan_dft_r2c_2d(n, n,
         rdata1.data(), reinterpret_cast<fftwf_complex*>(kdata1.data()), FFTW_ESTIMATE);
 
     // Fill & Execute
     fill_array(rdata1);
     fftwf_execute(plan1);
     fftwf_destroy_plan(plan1);
 
     // Validate
     std::cout << ">>> Out of place FFT "
               << (validate(rdata1,kdata1)?"match":"MISMATCH")
               << std::endl;
 
     //=============================================================
     // 2) CPU In-place 2D R2C (FFTW)
     //=============================================================
     Array<float,2> raw_data2(n,2*(n/2+1)); // padded
     Array<float,2> rdata2 = raw_data2(Range(0,n-1),Range(0,n-1));
 
     fftwf_plan plan2  = fftwf_plan_dft_r2c_2d(n, n,
         rdata2.data(), reinterpret_cast<fftwf_complex*>(rdata2.data()), FFTW_ESTIMATE);
     fill_array(rdata2);
     fftwf_execute(plan2);
     fftwf_destroy_plan(plan2);
 
     // Interpret the result
     Array<complex<float>, 2> kdata2(reinterpret_cast<complex<float>*>(rdata2.data()),
                                     shape(n, n/2 + 1),
                                     neverDeleteData);
 
     // Validate against the original rdata1
     std::cout << ">>> In-place FFT "
               << (validate(rdata1,kdata2)?"match":"MISMATCH")
               << std::endl;
 
     //=============================================================
     // 3) Simple GPU copy test
     //=============================================================
     Array<float,2> raw_data3(n, 2*(n/2 + 1));
     Array<float,2> data3 = raw_data3(Range(0,n-1), Range(0,n-1));
     fill_array(data3);
 
     Array<float,2> raw_data4(n, 2*(n/2 + 1));
     Array<float,2> data4 = raw_data4(Range(0,n-1), Range(0,n-1));
 
     size_t size_in_bytes = raw_data3.size() * sizeof(float);
     void *device_data;
     cudaMalloc(&device_data, size_in_bytes);
 
     // Host->Device
     cudaMemcpy(device_data, raw_data3.data(), size_in_bytes, cudaMemcpyHostToDevice);
     // Device->Host
     cudaMemcpy(raw_data4.data(), device_data, size_in_bytes, cudaMemcpyDeviceToHost);
 
     cudaFree(device_data);
 
     bool gpu_copy_ok = all(abs(data3 - data4) < 1e-5f);
     std::cout << ">>> GPU copy test " << (gpu_copy_ok ? "match" : "MISMATCH") << std::endl;
     
     //=============================================================
     // 4) GPU In-place 2D R2C with cufftPlanMany (Ex4)
     //=============================================================
     Array<float,2> raw_data5(n, 2*(n/2 + 1));
     Array<float,2> data5 = raw_data5(Range(0,n-1),Range(0,n-1));
     fill_array(data5);
 
     // Keep a copy for validation
     Array<float, 2> data5_copy(data5.copy());
 
     // Allocate on GPU
     float* rdata5;
     size_t size_bytes = raw_data5.size() * sizeof(float);
     cudaMalloc(&rdata5, size_bytes);
     cudaMemcpy(rdata5, raw_data5.data(), size_bytes, cudaMemcpyHostToDevice);
 
     // Instead of cufftPlan2d, use cufftPlanMany
     cufftHandle plan5;
     {
         int rank = 2;          // 2D transform
         int nembed[2] = {n, 2*(n/2+1)};  // Real data shape in memory
         int stride_in = 1; 
         int dist_in   = nembed[0]*nembed[1]; // for a single batch, or we can set 0
 
         int onembed[2] = {n, (n/2+1)};  // Complex data shape
         int stride_out = 1;
         int dist_out   = onembed[0]*onembed[1]; // single batch
 
         int batches = 1;
 
         int n_dims[2] = {n, n}; // actual NxN transform
 
         cufftResult ret = cufftPlanMany(&plan5,
                             rank,
                             n_dims,
                             // input layout
                             nembed, stride_in, dist_in,
                             // output layout
                             onembed, stride_out, dist_out,
                             CUFFT_R2C, 
                             batches);
         if(ret != CUFFT_SUCCESS){
             std::cerr<<"cufftPlanMany failed with error code "<<ret<<"\n";
             return 1;
         }
     }
 
     // Execute in-place R2C
     cufftExecR2C(plan5,
         reinterpret_cast<cufftReal*>(rdata5),
         reinterpret_cast<cufftComplex*>(rdata5));
     cufftDestroy(plan5);
 
     // Copy back
     cudaMemcpy(raw_data5.data(), rdata5, size_bytes, cudaMemcpyDeviceToHost);
     cudaFree(rdata5);
 
     // Interpret result as complex
     Array<complex<float>, 2> kdata5(
         reinterpret_cast<complex<float>*>(raw_data5.data()),
         shape(n, n/2 + 1),
         neverDeleteData
     );
 
     std::cout << ">>> GPU in-place FFT (PlanMany) "
               << (validate(data5_copy, kdata5) ? "match" : "MISMATCH")
               << std::endl;
  
     return 0;
 }
